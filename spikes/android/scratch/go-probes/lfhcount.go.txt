package main
import ("archive/zip";"fmt";"os")
func walkLFH(b []byte, cdOff int) (n int, pos int){
  pos=0
  for pos+4<=cdOff {
    if !(b[pos]==0x50&&b[pos+1]==0x4b&&b[pos+2]==0x03&&b[pos+3]==0x04){ return n,pos }
    csize:=int(uint32(b[pos+18])|uint32(b[pos+19])<<8|uint32(b[pos+20])<<16|uint32(b[pos+21])<<24)
    nlen:=int(b[pos+26])|int(b[pos+27])<<8
    elen:=int(b[pos+28])|int(b[pos+29])<<8
    pos=pos+30+nlen+elen+csize
    n++
  }
  return n,pos
}
func main(){
  for _,p:=range os.Args[1:]{
    b,err:=os.ReadFile(p); if err!=nil{panic(err)}
    eocd:=-1
    for i:=len(b)-22; i>=0 && i>len(b)-65558; i--{
      if b[i]==0x50&&b[i+1]==0x4b&&b[i+2]==0x05&&b[i+3]==0x06 { eocd=i; break }
    }
    cdOff:=int(uint32(b[eocd+16])|uint32(b[eocd+17])<<8|uint32(b[eocd+18])<<16|uint32(b[eocd+19])<<24)
    cdSize:=int(uint32(b[eocd+12])|uint32(b[eocd+13])<<8|uint32(b[eocd+14])<<16|uint32(b[eocd+15])<<24)
    // central dir entry count from EOCD
    cdCount:=int(uint16(b[eocd+10])|uint16(b[eocd+11])<<8)
    n,pos:=walkLFH(b,cdOff)
    r,err:=zip.OpenReader(p); if err!=nil{panic(err)}
    fmt.Printf("=== %s ===\n",p)
    fmt.Printf("  fileSize=%d  EOCD cdOff=%d cdSize=%d cdCountFromEOCD=%d\n",len(b),cdOff,cdSize,cdCount)
    fmt.Printf("  zip.OpenReader entries=%d   LFH-walk entries=%d  walkEnd=%d\n",len(r.File),n,pos)
    fmt.Printf("  => dataRegionBytes=%d  (LFH walk stops here, sig block begins)\n",pos)
    r.Close()
  }
}
