package main
import ("archive/zip";"fmt";"os")
func main(){
  for _,p:=range os.Args[1:]{
    r,err:=zip.OpenReader(p); if err!=nil{panic(err)}
    st,_:=os.Stat(p)
    var totComp uint64
    for _,f:=range r.File{ totComp+=f.CompressedSize64 }
    r.Close()
    b,err:=os.ReadFile(p); if err!=nil{panic(err)}
    eocd:=-1
    for i:=len(b)-22; i>=0 && i>len(b)-65558; i--{
      if b[i]==0x50&&b[i+1]==0x4b&&b[i+2]==0x05&&b[i+3]==0x06 { eocd=i; break }
    }
    fmt.Printf("=== %s ===\n  fileSize=%d sumCompressed=%d\n",p,st.Size(),totComp)
    if eocd>=0{
      cdSize:=int(uint32(b[eocd+12])|uint32(b[eocd+13])<<8|uint32(b[eocd+14])<<16|uint32(b[eocd+15])<<24)
      cdOff:=int(uint32(b[eocd+16])|uint32(b[eocd+17])<<8|uint32(b[eocd+18])<<16|uint32(b[eocd+19])<<24)
      fmt.Printf("  EOCD@%d cdOff=%d cdSize=%d\n",eocd,cdOff,cdSize)
      fmt.Printf("  APK signing block (cdOff - sumCompressed) = %d\n", cdOff-int(totComp))
      fmt.Printf("  tail (fileSize - cdOff - cdSize - 22) = %d\n", len(b)-cdOff-cdSize-22)
    } else { fmt.Println("  EOCD not found") }
  }
}
