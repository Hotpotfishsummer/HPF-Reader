package main
import ("archive/zip";"fmt";"os")
func main(){
  for _,p:=range os.Args[1:]{
    r,err:=zip.OpenReader(p); if err!=nil{panic(err)}
    st,_:=os.Stat(p)
    var totComp uint64
    for _,f:=range r.File{ totComp+=f.CompressedSize64 }
    fmt.Printf("=== %s ===\n  fileSize=%d entries=%d sumCompressed=%d\n",p,st.Size(),len(r.File),totComp)
    r.Close()
    // re-open raw and read EOCD to get central dir offset
    b,err:=os.ReadFile(p); if err!=nil{panic(err)}
    // EOCD signature 0x06054b50, search from end
    var eocd int = -1
    for i:=len(b)-22; i>=0 && i>len(b)-65558; i--{
      if b[i]==0x50&&b[i+1]==0x4b&&b[i+2]==0x05&&b[i+3]==0x06 { eocd=i; break }
    }
    if eocd>=0{
      cdSize:=uint32(b[eocd+12])|uint32(b[eocd+13])<<8|uint32(b[eocd+14])<<16|uint32(b[eocd+15])<<24
      cdOff:=uint32(b[eocd+16])|uint32(b[eocd+17])<<8|uint32(b[eocd+18])<<16|uint32(b[eocd+19])<<24
      fmt.Printf("  EOCD@%d centralDirOffset=%d centralDirSize=%d\n",eocd,cdOff,cdSize)
      fmt.Printf("  => data region bytes = %d ; central dir end = %d ; file end = %d\n",cdOff,cdOff+int(cdSize),len(b))
      fmt.Printf("  => APK signing block size approx = %d\n", int(cdOff)-int(totComp))
    } else { fmt.Println("  EOCD not found") }
  }
}
