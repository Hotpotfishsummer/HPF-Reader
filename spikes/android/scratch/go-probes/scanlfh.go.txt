package main
import ("fmt";"os";"sort")
func main(){
  p:=os.Args[1]
  b,err:=os.ReadFile(p); if err!=nil{panic(err)}
  eocd:=-1
  for i:=len(b)-22; i>=0 && i>len(b)-65558; i--{
    if b[i]==0x50&&b[i+1]==0x4b&&b[i+2]==0x05&&b[i+3]==0x06 { eocd=i; break }
  }
  cdOff:=int(uint32(b[eocd+16])|uint32(b[eocd+17])<<8|uint32(b[eocd+18])<<16|uint32(b[eocd+19])<<24)
  // walk local headers sequentially through data region
  pos:=0; n:=0
  type item struct{name string; start,end int}
  var items []item
  for pos+4<=cdOff {
    if !(b[pos]==0x50&&b[pos+1]==0x4b&&b[pos+2]==0x03&&b[pos+3]==0x04){ fmt.Printf("STOP: no LFH at %d\n",pos); break }
    csize:=int(uint32(b[pos+18])|uint32(b[pos+19])<<8|uint32(b[pos+20])<<16|uint32(b[pos+21])<<24)
    usize:=int(uint32(b[pos+22])|uint32(b[pos+23])<<8|uint32(b[pos+24])<<16|uint32(b[pos+25])<<24)
    nlen:=int(b[pos+26])|int(b[pos+27])<<8
    elen:=int(b[pos+28])|int(b[pos+29])<<8
    name:=string(b[pos+30:pos+30+nlen])
    dataStart:=pos+30+nlen+elen
    end:=dataStart+csize
    _=usize
    items=append(items,item{name,pos,end})
    n++
    pos=end
  }
  fmt.Printf("file=%s cdOff=%d walkedEntries=%d finalPos=%d\n",p,cdOff,n,pos)
  fmt.Printf("sumOfSpans=%d  leftover=%d\n",pos,cdOff-pos)
  sort.Slice(items,func(i,j int)bool{return (items[i].end-items[i].start)>(items[j].end-items[j].start)})
  fmt.Println("-- top 6 by raw span --")
  for i:=0;i<6&&i<len(items);i++{ fmt.Printf("   %-52s span=%d\n",items[i].name,items[i].end-items[i].start) }
}
