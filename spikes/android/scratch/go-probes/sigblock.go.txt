package main
import ("fmt";"os")
func main(){
  p:=os.Args[1]
  b,err:=os.ReadFile(p); if err!=nil{panic(err)}
  eocd:=-1
  for i:=len(b)-22; i>=0 && i>len(b)-65558; i--{
    if b[i]==0x50&&b[i+1]==0x4b&&b[i+2]==0x05&&b[i+3]==0x06 { eocd=i; break }
  }
  cdOff:=int(uint32(b[eocd+16])|uint32(b[eocd+17])<<8|uint32(b[eocd+18])<<16|uint32(b[eocd+19])<<24)
  fmt.Printf("cdOff=%d\n",cdOff)
  // APK Signing Block magic is 16 bytes right before cdOff
  magicStart:=cdOff-16
  fmt.Printf("bytes[%d:%d]=%q\n",magicStart,cdOff,string(b[magicStart:cdOff]))
  // size field is 8 bytes before magic
  if magicStart-8>=0 {
    sz:=uint64(b[magicStart-8])|uint64(b[magicStart-7])<<8|uint64(b[magicStart-6])<<16|uint64(b[magicStart-5])<<24|uint64(b[magicStart-4])<<32|uint64(b[magicStart-3])<<40|uint64(b[magicStart-2])<<48|uint64(b[magicStart-1])<<56
    fmt.Printf("sigBlockSizeField=%d -> block starts at %d\n",sz,int64(magicStart-8)-int64(sz))
  }
  // scan whole file for 'APK Sig Block 42' occurrences
  for i:=0;i+16<=len(b);i++{
    if string(b[i:i+16])=="APK Sig Block 42"{ fmt.Printf("found magic at %d\n",i) }
  }
}
