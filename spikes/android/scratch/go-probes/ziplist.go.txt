package main
import ("archive/zip";"fmt";"os";"strings")
func main(){
  for _,p:=range os.Args[1:]{
    r,err:=zip.OpenReader(p); if err!=nil{fmt.Println(p,"ERR",err);continue}
    var tot uint64
    fmt.Println("=== "+p+" ===")
    for _,f:=range r.File{
      tot+=f.CompressedSize64
      if strings.HasPrefix(f.Name,"lib/")||strings.HasPrefix(f.Name,"assets/frontend/dist/assets/index-")||strings.Contains(f.Name,"sample.epub"){
        fmt.Printf("  %-46s comp=%10d uncomp=%10d\n",f.Name,f.CompressedSize64,f.UncompressedSize64)
      }
    }
    st,_:=os.Stat(p)
    fmt.Printf("  entries=%d  sum(comp)=%d  fileSize=%d\n",len(r.File),tot,st.Size())
    r.Close()
  }
}
